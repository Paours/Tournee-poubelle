function startTracking() { if (!navigator.geolocation) { alert("Votre navigateur ne supporte pas la géolocalisation."); return; } document.getElementById('startBtn').disabled = true; document.getElementById('stopBtn').disabled = false; document.getElementById('addBinBtn').disabled = false; document.getElementById('saveBtn').disabled = true; currentPath = []; currentBins = []; lastKnownPos = null; clearMapMarkers(); polyline.setLatLngs([]); if (decorator) decorator.setPaths(polyline); // Tentative de récupération immédiate navigator.geolocation.getCurrentPosition( function(pos) { updatePosition(pos); }, function(err) { console.warn("Erreur GPS initiale :", err); if (err.code === 1) { alert("Accès GPS refusé. Veuillez autoriser la géolocalisation dans votre navigateur."); } }, { enableHighAccuracy: false, timeout: 10000, maximumAge: 0 } ); // Suivi en continu watchId = navigator.geolocation.watchPosition( function(pos) { updatePosition(pos); }, function(err) { console.warn("Erreur suivi GPS :", err); }, { enableHighAccuracy: true, timeout: 15000, maximumAge: 5000 } ); } function updatePosition(pos) { var lat = pos.coords.latitude; var lon = pos.coords.longitude; lastKnownPos = { lat: lat, lon: lon, time: new Date().toISOString() }; currentPath.push(lastKnownPos); polyline.addLatLng([lat, lon]); if (decorator) decorator.setPaths(polyline); if (!currentPositionMarker) { currentPositionMarker = L.marker([lat, lon]).addTo(map); map.setView([lat, lon], 17); } else { currentPositionMarker.setLatLng([lat, lon]); map.panTo([lat, lon]); } } function addBinStop() { // Si le GPS ne répond toujours pas, on prend le centre actuel de la carte var targetLat, targetLon; if (lastKnownPos) { targetLat = lastKnownPos.lat; targetLon = lastKnownPos.lon; } else { var center = map.getCenter(); targetLat = center.lat; targetLon = center.lng; } var binNumber = currentBins.length + 1; var note = prompt("Nom de l'arrêt / Poubelle :", "Arrêt N°" + binNumber); if (note === null) return; var binPoint = { number: binNumber, name: note || ("Arrêt N°" + binNumber), lat: targetLat, lon: targetLon, time: new Date().toLocaleTimeString() }; currentBins.push(binPoint); addBinMarkerToMap(binPoint); }
